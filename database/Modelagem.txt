CREATE DATABASE evonium;
GO

USE evonium;
GO

-- =========================================
-- TABELA DE USUÁRIOS
-- =========================================
CREATE TABLE tab_usuarios(
	id_usuario INT PRIMARY KEY IDENTITY,

	login VARCHAR(30) NOT NULL UNIQUE,
	email VARCHAR(50) NOT NULL UNIQUE,
	senha VARCHAR(100) NOT NULL,
	telefone VARCHAR(15) NOT NULL,
	nivel INT NOT NULL
);


-- =========================================
-- TABELA DE FUNCIONÁRIOS
-- =========================================
CREATE TABLE tab_funcionarios(
	id_funcionario INT PRIMARY KEY IDENTITY,

	id_usuario INT NOT NULL UNIQUE,

	nome VARCHAR(50) NOT NULL,
	email VARCHAR(50) NOT NULL,
	cpf VARCHAR(14) NOT NULL UNIQUE,
	rg VARCHAR(12) NOT NULL UNIQUE,
	telefone VARCHAR(20) NOT NULL,

	CONSTRAINT fk_func_usuario
	FOREIGN KEY (id_usuario)
	REFERENCES tab_usuarios(id_usuario)
);


-- =========================================
-- TABELA DE CLIENTES
-- =========================================
CREATE TABLE tab_clientes(
	id_cliente INT PRIMARY KEY IDENTITY,
	id_usuario INT NOT NULL UNIQUE,

	nome VARCHAR(50) NOT NULL,
	email VARCHAR(50) NOT NULL UNIQUE,
	fone VARCHAR(15) NOT NULL,
	dtnasc DATE NOT NULL,
	sexo VARCHAR(10) NOT NULL,

	-- 1 = ativo
	-- 0 = inativo
	status BOOLEAN

	CONSTRAINT fk_cliente_usuario 
	FOREIGN KEY (id_usuario) 
	REFERENCES tab_usuarios(id_usuario)
);


-- =========================================
-- TABELA DE ENDEREÇOS
-- =========================================
CREATE TABLE tab_enderecos(
	id_endereco INT PRIMARY KEY IDENTITY,

	id_cliente INT NOT NULL,

	rua VARCHAR(100) NOT NULL,
	numero VARCHAR(10) NOT NULL,
	bairro VARCHAR(50) NOT NULL,
	cidade VARCHAR(50) NOT NULL,
	estado CHAR(2) NOT NULL,
	cep VARCHAR(10) NOT NULL,

	CONSTRAINT fk_end_cliente
	FOREIGN KEY (id_cliente)
	REFERENCES tab_clientes(id_cliente)
);


-- =========================================
-- TABELA DE CATEGORIAS
-- 1 categoria -> N produtos
-- =========================================
CREATE TABLE tab_categorias(
	id_categoria INT PRIMARY KEY IDENTITY,

	nome VARCHAR(30) NOT NULL UNIQUE
);


-- =========================================
-- TABELA DE PRODUTOS
-- =========================================
CREATE TABLE tab_produtos(
	id_produto INT PRIMARY KEY IDENTITY,

	id_categoria INT NOT NULL,

	nome VARCHAR(50) NOT NULL,
	genero VARCHAR(20) NOT NULL,
	tipo VARCHAR(20) NOT NULL,
	fragrancia VARCHAR(50) NOT NULL,

	volume_ml INT NOT NULL,

	preco DECIMAL(10,2) NOT NULL,

	data_validade DATE NOT NULL,

	data_cadastro DATE NOT NULL DEFAULT GETDATE(),

	CONSTRAINT fk_prod_categoria
	FOREIGN KEY (id_categoria)
	REFERENCES tab_categorias(id_categoria)
);


-- =========================================
-- TABELA DE ESTOQUE
-- =========================================
CREATE TABLE tab_estoque(
	id_estoque INT PRIMARY KEY IDENTITY,

	id_produto INT NOT NULL UNIQUE,

	qtde INT NOT NULL,

	CONSTRAINT fk_estoque_produto
	FOREIGN KEY (id_produto)
	REFERENCES tab_produtos(id_produto)
);


-- =========================================
-- TABELA DE PEDIDOS
-- =========================================
CREATE TABLE tab_pedidos(
	id_pedido INT PRIMARY KEY IDENTITY,

	id_cliente INT NOT NULL,
	id_usuario INT NOT NULL,
	id_funcionario INT NULL,

	data DATETIME NOT NULL DEFAULT GETDATE(),

	total DECIMAL(10,2) NOT NULL,

	CONSTRAINT fk_pedido_cliente
	FOREIGN KEY (id_cliente)
	REFERENCES tab_clientes(id_cliente),

	CONSTRAINT fk_pedido_usuario
	FOREIGN KEY (id_usuario)
	REFERENCES tab_usuarios(id_usuario),

	CONSTRAINT fk_pedido_funcionario
	FOREIGN KEY (id_funcionario)
	REFERENCES tab_funcionarios(id_funcionario)
);


-- =========================================
-- TABELA DE ITENS DO PEDIDO
-- Resolve N:N entre pedidos e produtos
-- =========================================
CREATE TABLE tab_itens_pedido(
	id_item INT PRIMARY KEY IDENTITY,

	id_pedido INT NOT NULL,
	id_produto INT NOT NULL,

	valor DECIMAL(10,2) NOT NULL,

	qtde INT NOT NULL,

	subtotal DECIMAL(10,2) NOT NULL,

	CONSTRAINT fk_item_pedido
	FOREIGN KEY (id_pedido)
	REFERENCES tab_pedidos(id_pedido),

	CONSTRAINT fk_item_produto
	FOREIGN KEY (id_produto)
	REFERENCES tab_produtos(id_produto)
);


CARDINALIDADES:
USUÁRIOS -> FUNCIONÁRIOS
1 : 1

CLIENTES -> ENDEREÇOS
1 : N

CLIENTES -> PEDIDOS
1 : N

USUÁRIOS -> PEDIDOS
1 : N

FUNCIONÁRIOS -> PEDIDOS
1 : N

CATEGORIAS -> PRODUTOS
1 : N

PRODUTOS -> ESTOQUE
1 : 1

PEDIDOS -> ITENS_PEDIDO
1 : N

PRODUTOS -> ITENS_PEDIDO
1 : N

CLIENTES -> USUÁRIOS
1 : 1
